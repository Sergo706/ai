This page brings the tool subsystem pieces together into two complete recipes:

1. A **production custom tool** combining Zod validation, `toolError()`, `serializeForModel: jsonToMarkdown`, `withToolCache` + `normalizeArgs`, live sub-progress via `context.emit`, and `limitTool`/`trackTool`.
2. A **standalone low-level AI client** combining `executeToolLoop()`, `createMcpToolset()`, `resolveCommonTools()`, and `buildToolRegistry()` against any OpenAI-compatible endpoint (such as OpenRouter or Ollama) using an in-memory SQLite + memory cache setup (zero external Postgres/Redis servers and zero disk files).

## Build a production custom tool

Whether you register a tool on `createAgent({ tools: { specs } })` or compose wrappers manually with `buildToolRegistry()`, this pattern covers every production concern:

- **Input validation with Zod + `toolError()`**: returns structured JSON error envelopes that the LLM can self-correct on and that `withToolCache()` automatically refuses to cache.
- **Compact model wire format (`serializeForModel: jsonToMarkdown`)**: keeps raw typed objects in your trace/host code while rendering compact Markdown tables for the LLM context window.
- **Argument-normalized caching (`cache: { ttl, scoped, normalizeArgs }`)**: prevents cache misses from case differences, whitespace, or non-semantic arguments.
- **Live sub-progress (`context.emit`)**: emits `agent_progress` or custom `workflow_progress` updates during slow multi-step tool calls while honoring `context.signal` for cancellation.
- **Call caps and concurrency (`maxCalls`, `concurrency`)**: prevents runaway model loops and caps parallel external requests.

```ts [src/tools/competitorPricingTool.ts]
import {
  getConfiguration,
  jsonToMarkdown,
  toolError,
  type ToolHandler,
  type ToolSpec,
} from "@riavzon/ai";
import { z } from "zod";

export interface PricingMeta {
  tenantId: string;
  pricingApiKey: string;
}

export interface PricingToolDeps {
  tenantId: string;
  pricingApiKey: string;
}

const pricingInputSchema = z.strictObject({
  sku: z.string().trim().min(2).transform((s) => s.toUpperCase()),
  region: z.enum(["US", "EU", "UK"]).default("US"),
  // Stripped from the cache key because it is only for audit logging
  requestNote: z.string().optional(),
});

export function lookupCompetitorPricing(deps: PricingToolDeps): ToolHandler {
  const log = getConfiguration().logger;

  return async (rawArgs, execCtx) => {
    log.debug({ rawArgs, tenantId: deps.tenantId }, "agent ran tool lookup_competitor_pricing");

    const parsed = pricingInputSchema.safeParse(rawArgs);
    if (!parsed.success) {
      return toolError(
        "VALIDATION_ERROR",
        parsed.error.issues.map((i) => `${i.path.join(".")}: ${i.message}`).join("; "),
      );
    }

    const { sku, region } = parsed.data;
    if (!deps.pricingApiKey) {
      return toolError("MISCONFIGURED", "Pricing API credentials are missing.");
    }

    // Stream optional sub-progress frames while waiting on upstream APIs
    execCtx.emit?.({
      type: "agent_progress",
      agent: "pricing_fetcher",
      state: "running",
    });

    try {
      const response = await fetch(
        `https://pricing.example.com/v1/quotes?sku=${encodeURIComponent(sku)}&region=${region}`,
        {
          headers: {
            Authorization: `Bearer ${deps.pricingApiKey}`,
            "X-Tenant-Id": deps.tenantId,
          },
          signal: execCtx.signal,
        },
      );

      if (!response.ok) {
        execCtx.emit?.({ type: "agent_progress", agent: "pricing_fetcher", state: "failed" });
        return toolError("UPSTREAM_ERROR", `Pricing API returned HTTP ${response.status}`);
      }

      const data = (await response.json()) as {
        sku: string;
        currency: string;
        offers: { vendor: string; price: number; inStock: boolean }[];
      };

      execCtx.emit?.({ type: "agent_progress", agent: "pricing_fetcher", state: "done" });
      return data;
    } catch (err) {
      execCtx.emit?.({ type: "agent_progress", agent: "pricing_fetcher", state: "failed" });
      if (execCtx.signal?.aborted) {
        return toolError("ABORTED", "Pricing lookup was cancelled.");
      }
      return toolError("NETWORK_ERROR", err instanceof Error ? err.message : "Request failed");
    }
  };
}

export const competitorPricingTool: ToolSpec<PricingMeta> = {
  name: "lookup_competitor_pricing",
  definitions: {
    type: "function",
    function: {
      name: "lookup_competitor_pricing",
      description: "Fetch live competitor price quotes for a SKU in a region.",
      parameters: z.toJSONSchema(pricingInputSchema, { io: "input" }),
    },
  },
  maxCalls: 3,
  concurrency: 2,
  cache: {
    ttl: 600,
    scoped: true,
    // Normalize through Zod defaults/transforms and drop volatile audit fields
    normalizeArgs: (args) => {
      const parsed = pricingInputSchema.safeParse(args);
      if (!parsed.success) return args;
      const { sku, region } = parsed.data;
      return { sku, region };
    },
  },
  stream: {
    emitArgs: true,
    emitResult: "preview",
    // Model receives compact Markdown; TraceCollector still stores the raw object
    serializeForModel: (result) =>
      jsonToMarkdown(result, { headingLevel: 3, maxStringLength: 500 }),
  },
  make: (ctx) =>
    lookupCompetitorPricing({
      tenantId: ctx.meta?.tenantId ?? "",
      pricingApiKey: ctx.meta?.pricingApiKey ?? "",
    }),
};
```

### Composing wrappers manually without `buildToolRegistry()`

If you run your own executor map instead of `ToolSpec[]`, compose the exact same middleware chain manually using `limitTool()`, `withToolCache()`, and `trackTool()`:

```ts [src/tools/manualWrappers.ts]
import {
  limitTool,
  trackTool,
  withToolCache,
  type ToolHandler,
} from "@riavzon/ai";

export function wrapCustomHandler(
  name: string,
  baseHandler: ToolHandler,
  opts: {
    userId: string;
    maxCalls: number;
    ttlSeconds: number;
    counts: Map<string, number>;
    usedTools: Set<string>;
  },
): ToolHandler {
  // Match buildToolRegistry()'s exact wrapping order: limit -> cache -> track
  const capped = limitTool(name, opts.maxCalls, baseHandler, opts.counts);
  const cached = withToolCache(
    name,
    capped,
    opts.ttlSeconds,
    opts.userId,
    (args) => ({ sku: String(args.sku ?? "").trim().toUpperCase() }),
  );
  return trackTool(name, cached, opts.usedTools);
}
```

::note
`withToolCache()`, `executeToolLoop()`, `createMcpToolset()`, and built-in tools read logger and storage settings from `configuration()`. For lightweight scripts with no external database or Redis server, pass `db: ":memory:"` and `redis: "memory"` to `configuration()`.
::

## Build a standalone low-level AI client

When you want full control over prompt assembly, provider clients (OpenRouter, Ollama, vLLM, or OpenAI), and streaming without `createAgent()`'s conversation/persistence layer, combine `resolveCommonTools()`, `createMcpToolset()`, `buildToolRegistry()`, and `executeToolLoop()`. Passing `db: ":memory:"` and `redis: "memory"` initializes the logger and cache in RAM with zero external infrastructure or disk files:

```ts [src/standalone-client.ts]
import OpenAI from "openai";
import {
  buildToolRegistry,
  configuration,
  createMcpToolset,
  executeToolLoop,
  jsonToMarkdown,
  resolveCommonTools,
  type ToolSpec,
} from "@riavzon/ai";

// Works with Ollama, OpenRouter, vLLM, or OpenAI
const client = new OpenAI({
  apiKey: process.env.LLM_API_KEY ?? "ollama",
  baseURL: process.env.LLM_BASE_URL ?? "http://localhost:11434/v1",
});

// Zero-infrastructure init: in-memory SQLite + in-memory cache (no Postgres, no Redis, no files)
await configuration({
  openAiConfig: client,
  embeddingConfig: {
    retrievalModel: "text-embedding-3-small",
    matchingModel: "text-embedding-3-small",
    dimensions: 1536,
  },
  db: ":memory:",
  redis: "memory",
});

// 1. Custom local tool
const readEnvInfo: ToolSpec = {
  name: "runtime_info",
  definitions: {
    type: "function",
    function: {
      name: "runtime_info",
      description: "Return Node.js version and current UTC timestamp.",
      parameters: { type: "object", properties: {} },
    },
  },
  stream: {
    serializeForModel: (res) => jsonToMarkdown(res),
  },
  make: () => async () => ({
    node: process.version,
    platform: process.platform,
    nowUtc: new Date().toISOString(),
  }),
};

// 2. Connect to optional remote MCP servers
const mcp = await createMcpToolset(
  process.env.MCP_URL
    ? [
        {
          namespace: "docs",
          url: process.env.MCP_URL,
          auth: process.env.MCP_TOKEN
            ? { type: "bearer", token: process.env.MCP_TOKEN }
            : undefined,
          policy: { concurrency: 2, maxCalls: 5 },
        },
      ]
    : [],
);

try {
  // 3. Merge built-in common tools + custom specs + MCP specs in one registry
  const specs = [
    ...resolveCommonTools(["calculator", "get_current_datetime"]),
    readEnvInfo,
    ...mcp.specs,
  ];

  const registry = buildToolRegistry({ userId: "local-dev" }, specs);

  // 4. Drive executeToolLoop() directly with live event callbacks
  const result = await executeToolLoop({
    client,
    model: process.env.LLM_MODEL ?? "qwen2.5:14b",
    stream: true,
    messages: [
      { role: "system", content: "You are a helpful engineering assistant." },
      {
        role: "user",
        content: "What time is it in Tokyo, what is (42 * 19.5) + 100, and what Node version am I running?",
      },
    ],
    availableTools: Object.values(registry.definitions),
    handlers: registry.handlers,
    toolConcurrency: registry.toolConcurrency,
    toolStreaming: registry.toolStreaming,
    tools: { temperature: 0.1, maxTokens: 1024, maxRounds: 5, maxCalls: 10 },
    chat: { temperature: 0.3, maxTokens: 2048 },
    emit: (event) => {
      if (event.type === "tool_start") {
        process.stderr.write(`[calling ${event.name}]\n`);
      } else if (event.type === "chunk") {
        process.stdout.write(event.text);
      }
    },
  });

  process.stdout.write("\n");
  console.error("Tools used:", [...registry.usedTools]);
  console.error("Final answer length:", result.content.length);
} finally {
  // Always close MCP sessions when the script exits
  await mcp.close();
}
```
