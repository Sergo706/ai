This page walks through complete, copy-pasteable integrations that connect every major `@riavzon/ai` capability end-to-end: a fully configured production agent, an interactive terminal CLI consumer, HTTP server adapters for both Web Standard and Node.js runtimes, and a typed frontend SSE client with cursor tracking, reconnection, Stop, and Retry.

## Build a fully configured production agent

The example below wires together every subsystem available on `createAgent()`:

- **Provider transport & body options** via `model.request` and `model.body`
- **Built-in tools with overrides** (`calculator`, `get_current_datetime`, tenant-scoped `search_db`, `get_document`, and `pdf_creator` with a custom download route)
- **Custom `ToolSpec<AppMeta>`** combining Zod validation, `toolError()`, `serializeForModel: jsonToMarkdown`, caching with `normalizeArgs`, concurrency limits, and live sub-progress via `context.emit`
- **Remote MCP server** with dynamic bearer auth, exclusions, and shared policy
- **Persistence, compaction, attachments, semantic cache, custom `TraceExtension`, and durable `runs`**
- **Graceful shutdown** calling `await agent.close()`

```ts [src/agent.ts]
import OpenAI from "openai";
import { Pool } from "pg";
import Redis from "ioredis";
import {
  configuration,
  createAgent,
  getConfiguration,
  jsonToMarkdown,
  pdfCreator,
  searchDb,
  startRuns,
  toolError,
  type AgentEvent,
  type ToolHandler,
  type ToolSpec,
  type TraceExtension,
} from "@riavzon/ai";
import { z } from "zod";

export interface AppMeta {
  orgId: string;
  region: string;
}

// 1. Custom ToolSpec backed by a reusable typed ToolHandler factory
const orderLookupInput = z.strictObject({
  orderId: z.string().trim().min(1).transform((id) => id.toUpperCase()),
  includeAuditTrail: z.boolean().default(false),
});

export function lookupOrder(deps: { orgId: string }): ToolHandler {
  const log = getConfiguration().logger;

  return async (rawArgs, execCtx) => {
    log.debug({ rawArgs, orgId: deps.orgId }, "agent ran tool get_order_status");

    const parsed = orderLookupInput.safeParse(rawArgs);
    if (!parsed.success) {
      return toolError("VALIDATION_ERROR", parsed.error.issues.map((i) => i.message).join("; "));
    }

    const res = await fetch(
      `https://orders.internal/${deps.orgId}/${encodeURIComponent(parsed.data.orderId)}?audit=${String(parsed.data.includeAuditTrail)}`,
      { signal: execCtx.signal },
    );

    if (!res.ok) {
      return toolError("ORDER_NOT_FOUND", `Order ${parsed.data.orderId} returned HTTP ${res.status}`);
    }

    return res.json();
  };
}

const getOrderStatus: ToolSpec<AppMeta> = {
  name: "get_order_status",
  definitions: {
    type: "function",
    function: {
      name: "get_order_status",
      description: "Fetch real-time fulfillment status for an order in the current organization.",
      parameters: z.toJSONSchema(orderLookupInput, { io: "input" }),
    },
  },
  maxCalls: 4,
  concurrency: 2,
  cache: {
    ttl: 120,
    scoped: true,
    normalizeArgs: (args) => {
      const parsed = orderLookupInput.safeParse(args);
      return parsed.success ? parsed.data : args;
    },
  },
  stream: {
    emitArgs: true,
    emitResult: "preview",
    // Keep structured JSON for the host/trace while sending compact Markdown to the LLM
    serializeForModel: (result) => jsonToMarkdown(result),
  },
  make: (ctx) => lookupOrder({ orgId: ctx.meta?.orgId ?? "default" }),
};

// 2. Custom TraceExtension recording wall-clock tool timings per turn
function createTimingExtension(): TraceExtension<{ toolCount: number; totalToolMs: number }> {
  let toolCount = 0;
  let totalToolMs = 0;

  return {
    apply(event: AgentEvent) {
      if (event.type === "tool_end") {
        toolCount += 1;
        totalToolMs += event.durationMs;
      }
    },
    toTrace() {
      return toolCount > 0 ? { toolCount, totalToolMs } : undefined;
    },
  };
}

// 3. Configure runtime once at boot
export async function bootSupportAgent() {
  const pool = new Pool({ connectionString: process.env.DATABASE_URL! });
  const redis = new Redis(process.env.REDIS_URL!);
  const openai = new OpenAI({
    apiKey: process.env.OPENAI_API_KEY!,
    baseURL: process.env.OPENAI_BASE_URL,
  });

  await configuration({
    openAiConfig: openai,
    embeddingConfig: {
      retrievalModel: "text-embedding-3-small",
      matchingModel: "text-embedding-3-small",
      dimensions: 1536,
    },
    db: pool,
    redis,
  });

  startRuns();

  const agent = await createAgent<AppMeta>({
    model: {
      model: "gpt-4o",
      temperature: 0.3,
      maxTokens: 4096,
      thinkingEffort: "medium",
      tools: {
        temperature: 0.1,
        maxTokens: 2048,
        maxRounds: 6,
        maxCalls: 15,
      },
      body: {
        parallel_tool_calls: true,
      },
      request: {
        timeout: 45_000,
        maxRetries: 2,
        headers: {
          "X-App-Name": "support-assistant",
        },
      },
    },
    systemInstructions: [
      "You are a factual support assistant.",
      "Always verify order status with get_order_status before making shipping commitments.",
      "Cite document titles when answering from search_db.",
    ].join("\n"),
    tools: {
      common: [
        "calculator",
        "get_current_datetime",
        {
          name: "search_db",
          cache: { ttl: 3600, scoped: true },
          make: (ctx) => searchDb(ctx.userId),
        },
        { name: "get_document", maxCalls: 5 },
        {
          name: "pdf_creator",
          make: () => pdfCreator("/api/downloads/files"),
        },
      ],
      specs: [getOrderStatus],
      mcp: [
        {
          namespace: "crm",
          url: process.env.CRM_MCP_URL!,
          auth: {
            type: "bearer",
            token: async () => process.env.CRM_MCP_TOKEN!,
          },
          exclude: ["delete_customer"],
          policy: {
            concurrency: 2,
            maxCalls: 5,
            cache: { ttl: 300, scoped: true },
          },
        },
      ],
    },
    conversation: { persist: true },
    compaction: {
      temperature: 0.2,
      maxTokens: 1200,
      tokenThreshold: 24_000,
      keepRecentMessages: 20,
      reserveTokens: 4000,
    },
    attachments: {
      maxFiles: 3,
      maxBytes: 10_000_000,
      maxPdfPages: 10,
    },
    semanticCache: {
      similarity: 0.92,
      windowHours: 24,
      // Never reuse answers that depended on live order or CRM data
      poisonTools: ["get_order_status", "crm__get_ticket"],
    },
    trace: [createTimingExtension()],
    runs: {
      kind: "chat",
      name: "support-agent",
    },
    meta: {
      orgId: "default-org",
      region: "us-east",
    },
  });

  // Clean up MCP connections on process exit
  const shutdown = async () => {
    await agent.close();
    process.exit(0);
  };
  process.once("SIGTERM", shutdown);
  process.once("SIGINT", shutdown);

  return agent;
}
```

::tip
Because `createAgent()` opens and maintains MCP sessions, always call `await agent.close()` during graceful shutdown so child connections and handles are released cleanly.

## Build an interactive CLI consumer

`agent.chat()` returns `{ stream, done }` where `stream` is a standard `AsyncIterable<SseEvent>`. You can consume it directly in a terminal CLI without any HTTP server — rendering live reasoning updates, tool progress indicators, streamed tokens, and cancelling mid-turn on `Ctrl+C`.

```ts [src/cli.ts]
import { createInterface } from "node:readline/promises";
import { stdin as input, stdout as output } from "node:process";
import OpenAI from "openai";
import { configuration, createAgent } from "@riavzon/ai";

await configuration({
  openAiConfig: new OpenAI({
    apiKey: process.env.OPENAI_API_KEY!,
    baseURL: process.env.OPENAI_BASE_URL,
  }),
  embeddingConfig: {
    retrievalModel: "text-embedding-3-small",
    matchingModel: "text-embedding-3-small",
    dimensions: 1536,
  },
  db: "./cli-chat.sqlite",
  redis: "memory",
});

const agent = await createAgent({
  model: { model: "gpt-4o-mini", thinkingEffort: "medium" },
  systemInstructions: "You are a concise terminal assistant.",
  tools: { common: ["calculator", "get_current_datetime"] },
  conversation: { persist: true },
});

const rl = createInterface({ input, output });
let conversationId: number | undefined;
let activeAbort: AbortController | null = null;

// Cancel the active turn on Ctrl+C without killing the CLI session
process.on("SIGINT", () => {
  if (activeAbort) {
    activeAbort.abort();
    process.stderr.write("\n[cancelled]\n");
    return;
  }
  rl.close();
  void agent.close().finally(() => process.exit(0));
});

try {
  while (true) {
    const line = (await rl.question("\nyou> ")).trim();
    if (!line || line === "/exit") break;

    activeAbort = new AbortController();
    const turn = agent.chat({
      userId: "cli-user",
      conversationId,
      message: line,
      signal: activeAbort.signal,
    });

    let printedHeader = false;
    let inReasoning = false;

    for await (const event of turn.stream) {
      switch (event.type) {
        case "init":
          conversationId = event.conversationId;
          break;
        case "reasoning":
          if (!inReasoning) {
            process.stderr.write("\x1b[2m[thinking] ");
            inReasoning = true;
          }
          process.stderr.write(event.delta);
          break;
        case "tool_start":
          if (inReasoning) {
            process.stderr.write("\x1b[0m\n");
            inReasoning = false;
          }
          process.stderr.write(
            `\x1b[36m[tool ${event.callIndex}/${event.callTotal}] ${event.name}...\x1b[0m\n`,
          );
          break;
        case "tool_end":
          process.stderr.write(
            `\x1b[32m[tool ${event.name}] done in ${event.durationMs}ms\x1b[0m\n`,
          );
          break;
        case "chunk":
          if (inReasoning) {
            process.stderr.write("\x1b[0m\n");
            inReasoning = false;
          }
          if (!printedHeader) {
            output.write("ai> ");
            printedHeader = true;
          }
          output.write(event.text);
          break;
        case "error":
          process.stderr.write(`\x1b[31m[error] ${event.message}\x1b[0m\n`);
          break;
        case "done":
          if (inReasoning) process.stderr.write("\x1b[0m");
          output.write("\n");
          break;
      }
    }

    const result = await turn.done;
    activeAbort = null;
    if (result.cached) {
      process.stderr.write("\x1b[2m(served from semantic cache)\x1b[0m\n");
    }
  }
} finally {
  rl.close();
  await agent.close();
}
```

## Expose streaming over HTTP servers

Convert `turn.stream` into a standard `ReadableStream<Uint8Array>` with `framesToStream()` and attach SSE headers with `setSseHeaders()`.

::tabs
  :::tabs-item{label="Web Standard (Next.js / Hono / Nitro)" icon="i-lucide-globe"}
  Works out of the box anywhere the Fetch API `Request`/`Response` contract is native (Next.js App Router, Hono, Nuxt Nitro, Bun, Deno):

  ```ts [server/api/chat.post.ts]
  import {
    framesToStream,
    setSseHeaders,
  } from "@riavzon/ai";
  import { supportAgent } from "../lib/agent";

  export async function POST(req: Request): Promise<Response> {
    const body = (await req.json()) as {
      userId: string;
      conversationId?: number;
      message: string;
      orgId: string;
    };

    // Mint runId up front so the client receives it immediately via X-Run-Id
    const runId = crypto.randomUUID();
    const controller = new AbortController();

    // Forward client disconnects ONLY for stateless/non-resumable turns.
    // For durable runs (agent configured with `runs`), do NOT abort on
    // req.signal so the run keeps finishing in the background for reattach!
    const isDurableRun = true;
    if (!isDurableRun) {
      req.signal.addEventListener("abort", () => controller.abort(), { once: true });
    }

    const turn = supportAgent.chat({
      userId: body.userId,
      conversationId: body.conversationId,
      message: body.message,
      runId,
      meta: { orgId: body.orgId, region: "us-east" },
      signal: controller.signal,
    });

    const sseStream = framesToStream(turn.stream, {
      heartbeatMs: 10_000,
      onCancel: () => {
        if (!isDurableRun) controller.abort();
      },
    });

    // Optional server-side logging when the turn settles
    void turn.done.then((res) => {
      if (res.error) console.error("Turn failed:", res.error);
    });

    return new Response(sseStream, {
      headers: {
        ...setSseHeaders(),
        "X-Run-Id": runId,
      },
    });
  }
  ```
  :::

  :::tabs-item{label="Node.js (Express / Fastify / node:http)" icon="i-lucide-server"}
  Pipe the `ReadableStream<Uint8Array>` chunk by chunk into `ServerResponse`, flushing headers immediately and detaching cleanly on `req.on("close")`:

  ```ts [server/routes/chat.ts]
  import type { IncomingMessage, ServerResponse } from "node:http";
  import { framesToStream, setSseHeaders } from "@riavzon/ai";
  import { supportAgent } from "../lib/agent";

  export async function handleChat(
    req: IncomingMessage,
    res: ServerResponse,
    body: { userId: string; conversationId?: number; message: string; orgId: string },
  ): Promise<void> {
    const runId = crypto.randomUUID();
    const controller = new AbortController();
    const isDurableRun = true;

    const turn = supportAgent.chat({
      userId: body.userId,
      conversationId: body.conversationId,
      message: body.message,
      runId,
      meta: { orgId: body.orgId, region: "us-east" },
      signal: controller.signal,
    });

    const headers = {
      ...setSseHeaders(),
      "X-Run-Id": runId,
    };
    res.writeHead(200, headers);
    res.flushHeaders();

    const reader = framesToStream(turn.stream, { heartbeatMs: 10_000 }).getReader();
    let clientGone = false;

    req.on("close", () => {
      clientGone = true;
      void reader.cancel();
      // Only abort the turn when runs are NOT enabled; durable runs stay alive for reattach
      if (!isDurableRun) {
        controller.abort();
      }
    });

    try {
      while (!clientGone) {
        const { value, done } = await reader.read();
        if (done) break;
        if (!res.write(value)) {
          await new Promise<void>((resolve) => res.once("drain", resolve));
        }
      }
    } finally {
      reader.releaseLock();
      if (!res.writableEnded) res.end();
    }
  }
  ```
  :::

### Reattach, Stop, and Retry companion routes

Pair your chat endpoint with three thin run-control routes (full details in [Event Store & Reattach](/runs/event-store-and-reattach)):

```ts [server/api/runs.ts]
import {
  FIRST_ID,
  getRun,
  getRunById,
  requestAbort,
  retryRun,
  setSseHeaders,
  streamFromRun,
  streamFromStore,
} from "@riavzon/ai";

// GET /api/runs/:id/stream — resume from Last-Event-ID on any replica
export async function GET_stream(req: Request, runId: string, log: any): Promise<Response> {
  const since =
    req.headers.get("last-event-id") ??
    new URL(req.url).searchParams.get("since") ??
    FIRST_ID;

  const local = getRun(runId);
  if (local) {
    return new Response(
      streamFromRun({
        run: local,
        since,
        conversationId: local.conversationId,
        runId: local.id,
        log,
      }),
      { headers: setSseHeaders() },
    );
  }

  const row = await getRunById(runId);
  if (!row) return new Response("Run not found", { status: 404 });

  return new Response(
    streamFromStore({
      runId,
      since,
      conversationId: row.conversation_id ?? undefined,
      log,
      isRunAlive: async () => (await getRunById(runId))?.status === "running",
    }),
    { headers: setSseHeaders() },
  );
}

// POST /api/runs/:id/stop — abort locally + mark DB row so remote replicas stop on next heartbeat
export async function POST_stop(runId: string): Promise<Response> {
  getRun(runId)?.controller.abort();
  await requestAbort(runId);
  return Response.json({ ok: true });
}

// POST /api/runs/:id/retry — restart a failed, aborted, or interrupted run via its registered resumer
export async function POST_retry(runId: string): Promise<Response> {
  const outcome = await retryRun(runId);
  return Response.json(outcome, { status: outcome.ok ? 200 : 409 });
}
```

## Consume SSE streams in a browser or frontend client

Because `POST /api/chat` sends a JSON body (which browser `EventSource` does not support), parse SSE frames directly from `fetch().body`. The typed client below handles every `SseEvent` variant, tracks `Last-Event-ID`, automatically reattaches to `GET /api/runs/:id/stream` after network drops, and exposes `stop()` and `retry()` helpers.

```ts [client/aiStreamClient.ts]
import type { SseEvent, SseFrame } from "@riavzon/ai";

export type StreamFrame = SseEvent | SseFrame;

export interface ChatSessionState {
  conversationId?: number;
  runId?: string;
  lastEventId?: string;
  text: string;
  reasoning: string;
  activeTools: Map<string, { name: string; callIndex: number; callTotal: number }>;
  status: "idle" | "streaming" | "done" | "error";
  error?: string;
}

export interface StreamHandlers {
  onFrame: (frame: StreamFrame, eventId?: string) => void;
  onRunId?: (runId: string) => void;
}

async function consumeSseResponse(
  response: Response,
  handlers: StreamHandlers,
): Promise<{ sawTerminal: boolean; lastEventId?: string }> {
  if (!response.ok || !response.body) {
    throw new Error(`HTTP ${response.status}`);
  }

  const headerRunId = response.headers.get("X-Run-Id");
  if (headerRunId) handlers.onRunId?.(headerRunId);

  const reader = response.body.getReader();
  const decoder = new TextDecoder();
  let buffer = "";
  let sawTerminal = false;
  let lastEventId: string | undefined;

  while (true) {
    const { value, done } = await reader.read();
    if (done) break;

    buffer += decoder.decode(value, { stream: true });
    const blocks = buffer.split("\n\n");
    buffer = blocks.pop() ?? "";

    for (const block of blocks) {
      if (!block.trim() || block.startsWith(":")) continue; // Ignore heartbeats (": keep-alive")

      let id: string | undefined;
      const dataLines: string[] = [];

      for (const line of block.split("\n")) {
        if (line.startsWith("id:")) id = line.slice(3).trim();
        else if (line.startsWith("data:")) dataLines.push(line.slice(5).trim());
      }

      if (dataLines.length === 0) continue;
      if (id) lastEventId = id;

      const parsed = JSON.parse(dataLines.join("\n")) as StreamFrame;
      if (parsed.type === "done" || parsed.type === "error") {
        sawTerminal = true;
      }
      handlers.onFrame(parsed, id);
    }
  }

  return { sawTerminal, lastEventId };
}

export async function sendAndStreamChat(
  payload: { userId: string; orgId: string; conversationId?: number; message: string },
  onStateChange: (state: ChatSessionState) => void,
): Promise<ChatSessionState> {
  const state: ChatSessionState = {
    conversationId: payload.conversationId,
    text: "",
    reasoning: "",
    activeTools: new Map(),
    status: "streaming",
  };

  const applyFrame = (frame: StreamFrame, eventId?: string) => {
    if (eventId) state.lastEventId = eventId;

    switch (frame.type) {
      case "init":
        state.conversationId = frame.conversationId as number;
        if (typeof frame.runId === "string") state.runId = frame.runId;
        break;
      case "reasoning":
        state.reasoning += frame.delta as string;
        break;
      case "tool_start":
        state.activeTools.set(frame.id as string, {
          name: frame.name as string,
          callIndex: frame.callIndex as number,
          callTotal: frame.callTotal as number,
        });
        break;
      case "tool_end":
        state.activeTools.delete(frame.id as string);
        break;
      case "chunk":
        state.text += frame.text as string;
        break;
      case "workflow_start":
      case "workflow_progress":
      case "agent_progress":
        // Update workflow step indicator in your UI
        break;
      case "error":
        state.status = "error";
        state.error = frame.message as string;
        break;
      case "done":
        state.status = "done";
        break;
    }

    onStateChange({ ...state, activeTools: new Map(state.activeTools) });
  };

  // 1. Initial POST /api/chat stream
  try {
    const res = await fetch("/api/chat", {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify(payload),
    });
    const outcome = await consumeSseResponse(res, {
      onFrame: applyFrame,
      onRunId: (id) => {
        state.runId = id;
      },
    });
    if (outcome.sawTerminal) return state;
  } catch {
    // Fall through to reconnect if we already obtained a runId
  }

  // 2. Reattach seamlessly via Last-Event-ID when the connection drops mid-turn
  if (state.runId && state.status === "streaming") {
    const headers: Record<string, string> = {};
    if (state.lastEventId) headers["Last-Event-ID"] = state.lastEventId;

    const reconnect = await fetch(`/api/runs/${encodeURIComponent(state.runId)}/stream`, {
      headers,
    });
    await consumeSseResponse(reconnect, { onFrame: applyFrame });
  }

  return state;
}

export async function stopRun(runId: string): Promise<void> {
  await fetch(`/api/runs/${encodeURIComponent(runId)}/stop`, { method: "POST" });
}

export async function retryFailedRun(runId: string) {
  const res = await fetch(`/api/runs/${encodeURIComponent(runId)}/retry`, { method: "POST" });
  return res.json();
}
```
