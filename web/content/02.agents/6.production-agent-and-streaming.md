This page walks through complete, copy-pasteable integrations that connect every major `@riavzon/ai` capability end-to-end: a fully configured production agent, an interactive terminal CLI consumer, HTTP server adapters for both Web Standard and Node.js runtimes, and a typed frontend SSE client with cursor tracking, reconnection, Stop, and Retry.

## Build a fully configured production agent

The example below wires together every subsystem available on `createAgent()`:

- **Provider transport & body options** via `model.request` and `model.body`
- **Built-in tools with overrides** (`calculator`, `get_current_datetime`, tenant-scoped `search_db`, `get_document`, and `pdf_creator` with a custom download route)
- **Custom** **`ToolSpec<AppMeta>`** combining Zod validation, `toolError()`, `serializeForModel: jsonToMarkdown`, caching with `normalizeArgs`, concurrency limits, and cancellation via `execCtx.signal`
- **Remote MCP server** with dynamic bearer auth, exclusions, and shared policy
- **Persistence, compaction, attachments, semantic cache, and custom** **`TraceExtension`**
- **Graceful shutdown** calling `await agent.close()`

```ts [src/agent.ts]
import OpenAI from "openai";
import { Pool } from "pg";
import Redis from "ioredis";
import {
  configuration,
  createAgent,
  getConfiguration,
  jsonToMarkdown,
  pdfCreator,
  searchDb,
  startRuns,
  toolError,
  type AgentEvent,
  type ToolHandler,
  type ToolSpec,
  type TraceExtension,
} from "@riavzon/ai";
import { z } from "zod";

export interface AppMeta {
  orgId: string;
  region: string;
}

// 1. Custom ToolSpec backed by a reusable typed ToolHandler factory
const orderLookupInput = z.strictObject({
  orderId: z
    .string()
    .trim()
    .min(1)
    .transform((id) => id.toUpperCase()),
  includeAuditTrail: z.boolean().default(false),
});

export function lookupOrder(deps: { orgId: string }): ToolHandler {
  const log = getConfiguration().logger;

  return async (rawArgs, execCtx) => {
    log.debug(
      { rawArgs, orgId: deps.orgId },
      "agent ran tool get_order_status",
    );

    const parsed = orderLookupInput.safeParse(rawArgs);
    if (!parsed.success) {
      return toolError(
        "VALIDATION_ERROR",
        parsed.error.issues.map((i) => i.message).join("; "),
      );
    }

    const res = await fetch(
      `https://orders.internal/${deps.orgId}/${encodeURIComponent(parsed.data.orderId)}?audit=${String(parsed.data.includeAuditTrail)}`,
      { signal: execCtx.signal },
    );

    if (!res.ok) {
      return toolError(
        "ORDER_NOT_FOUND",
        `Order ${parsed.data.orderId} returned HTTP ${res.status}`,
      );
    }

    return res.json();
  };
}

const getOrderStatus: ToolSpec<AppMeta> = {
  name: "get_order_status",
  definitions: {
    type: "function",
    function: {
      name: "get_order_status",
      description:
        "Fetch real-time fulfillment status for an order in the current organization.",
      parameters: z.toJSONSchema(orderLookupInput, { io: "input" }),
    },
  },
  maxCalls: 4,
  concurrency: 2,
  cache: {
    ttl: 120,
    scoped: true,
    normalizeArgs: (args) => {
      const parsed = orderLookupInput.safeParse(args);
      return parsed.success ? parsed.data : args;
    },
  },
  stream: {
    emitArgs: true,
    emitResult: "preview",
    // Keep structured JSON for the host/trace while sending compact Markdown to the LLM
    serializeForModel: (result) => jsonToMarkdown(result),
  },
  make: (ctx) => lookupOrder({ orgId: ctx.meta?.orgId ?? "default" }),
};

// 2. Custom TraceExtension recording wall-clock tool timings per turn
function createTimingExtension(): TraceExtension<{
  toolCount: number;
  totalToolMs: number;
}> {
  let toolCount = 0;
  let totalToolMs = 0;

  return {
    apply(event: AgentEvent) {
      if (event.type === "tool_end") {
        toolCount += 1;
        totalToolMs += event.durationMs;
      }
    },
    toTrace() {
      return toolCount > 0 ? { toolCount, totalToolMs } : undefined;
    },
  };
}

// 3. Configure runtime once at boot
export async function bootSupportAgent() {
  const pool = new Pool({ connectionString: process.env.DATABASE_URL! });
  const redis = new Redis(process.env.REDIS_URL!);
  const openai = new OpenAI({
    apiKey: process.env.OPENAI_API_KEY!,
    baseURL: process.env.OPENAI_BASE_URL,
  });

  await configuration({
    openAiConfig: openai,
    embeddingConfig: {
      retrievalModel: "text-embedding-3-small",
      matchingModel: "text-embedding-3-small",
      dimensions: 1536,
    },
    db: pool,
    redis,
  });

  startRuns();

  const agent = await createAgent<AppMeta>({
    model: {
      model: "gpt-4o",
      temperature: 0.3,
      maxTokens: 4096,
      thinkingEffort: "medium",
      tools: {
        temperature: 0.1,
        maxTokens: 2048,
        maxRounds: 6,
        maxCalls: 15,
      },
      body: {
        parallel_tool_calls: true,
      },
      request: {
        timeout: 45_000,
        maxRetries: 2,
        headers: {
          "X-App-Name": "support-assistant",
        },
      },
    },
    systemInstructions: [
      "You are a factual support assistant.",
      "Always verify order status with get_order_status before making shipping commitments.",
      "Cite document titles when answering from search_db.",
    ].join("\n"),
    tools: {
      common: [
        "calculator",
        "get_current_datetime",
        {
          name: "search_db",
          cache: { ttl: 3600, scoped: true },
          make: (ctx) => searchDb(ctx.userId),
        },
        { name: "get_document", maxCalls: 5 },
        {
          name: "pdf_creator",
          make: () => pdfCreator("/api/downloads/files"),
        },
      ],
      specs: [getOrderStatus],
      mcp: [
        {
          namespace: "crm",
          url: process.env.CRM_MCP_URL!,
          auth: {
            type: "bearer",
            token: async () => process.env.CRM_MCP_TOKEN!,
          },
          exclude: ["delete_customer"],
          policy: {
            concurrency: 2,
            maxCalls: 5,
            cache: { ttl: 300, scoped: true },
          },
        },
      ],
    },
    conversation: { persist: true },
    compaction: {
      temperature: 0.2,
      maxTokens: 1200,
      tokenThreshold: 24_000,
      keepRecentMessages: 20,
      reserveTokens: 4000,
    },
    attachments: {
      maxFiles: 3,
      maxBytes: 10_000_000,
      maxPdfPages: 10,
    },
    semanticCache: {
      similarity: 0.92,
      windowHours: 24,
      // Never reuse answers that depended on live order or CRM data
      poisonTools: ["get_order_status", "crm__get_ticket"],
    },
    trace: [createTimingExtension()],
    runs: {
      kind: "chat",
      name: "support-agent",
    },
    meta: {
      orgId: "default-org",
      region: "us-east",
    },

  // Clean up MCP connections on process exit

    await agent.close();
    process.exit(0);

  process.once("SIGINT", shutdown);

  return agent;
}

export const supportAgent = await bootSupportAgent();
```

## Build an interactive CLI consumer

`await agent.chat()` returns a `Promise<ChatStream>` resolving to `{ frames, done }`, where `frames` is a standard `AsyncIterable<SseFrame>`. You can consume it directly in a terminal CLI without any HTTP server — rendering live reasoning updates, tool progress indicators, streamed tokens, and cancelling mid-turn on `Ctrl+C`.

```ts [src/cli.ts]
import { createInterface } from "node:readline/promises";
import { stdin as input, stdout as output } from "node:process";
import OpenAI from "openai";
import { configuration, createAgent, type Logger } from "@riavzon/ai";

// Silence internal runtime logs so stdout/stderr stay clean while streaming
const silentLogger: Logger = {
  debug: () => {},
  info: () => {},
  warn: () => {},
  error: () => {},
  child: () => silentLogger,
};

await configuration({
  openAiConfig: new OpenAI({
    apiKey: process.env.OPENAI_API_KEY!,
    baseURL: process.env.OPENAI_BASE_URL,
  }),
  embeddingConfig: {
    retrievalModel: "text-embedding-3-small",
    matchingModel: "text-embedding-3-small",
    dimensions: 1536,
    // Optional: point embeddings at a separate OpenAI-compatible client
    ...(process.env.EMBEDDING_BASE_URL
      ? {
          client: new OpenAI({
            apiKey:
              process.env.EMBEDDING_API_KEY ?? process.env.OPENAI_API_KEY!,
            baseURL: process.env.EMBEDDING_BASE_URL,
          }),
        }
      : {}),
  },
  db: "./cli-chat.sqlite",
  logger: silentLogger,
  redis: "memory",
});

const agent = await createAgent({
  model: { model: "gpt-4o-mini" },
  systemInstructions: "You are a concise, helpful terminal assistant.",
  tools: { common: ["calculator", "get_current_datetime", "extractor"] },
  conversation: { persist: true },
  semanticCache: {
    similarity: 0.9,
  },
});

const rl = createInterface({ input, output });
let conversationId: number | undefined;
let activeAbort: AbortController | null = null;

process.on("SIGINT", () => {
  if (activeAbort !== null) {
    activeAbort.abort();
    process.stderr.write("\n[cancelled]\n");
    return;
  }
  rl.close();
  void agent.close().finally(() => process.exit(0));
});

try {
  while (true) {
    const line = (await rl.question("\nyou> ")).trim();
    if (!line || line === "/exit") break;

    activeAbort = new AbortController();

    // `chat()` is async — it returns a Promise<ChatStream>.
    const turn = await agent.chat({
      userId: "cli-user",
      conversationId,
      message: line,
      signal: activeAbort.signal,
    });

    let printedHeader = false;
    let inReasoning = false;

    // The live iterator is `turn.frames`, not `turn.stream`.
    for await (const frame of turn.frames) {
      switch (frame.type) {
        case "init":
          if (typeof frame.conversationId === "number") {
            conversationId = frame.conversationId;
          }
          break;

        case "reasoning": {
          if (!inReasoning) {
            process.stderr.write("\n \x1b[2m[thinking] ");
            inReasoning = true;
          }
          process.stderr.write(String(frame.delta));
          break;
        }

        case "tool_start": {
          if (inReasoning) {
            process.stderr.write("\n \x1b[0m\n");
            inReasoning = false;
          }
          process.stderr.write(
            `\n \x1b[36m[tool ${String(frame.callIndex)}/${String(
              frame.callTotal,
            )}] ${String(frame.name)}...\x1b[0m\n`,
          );
          break;
        }

        case "tool_end": {
          process.stderr.write(
            `\n \x1b[32m[tool ${String(frame.name)}] done in ${String(
              frame.durationMs,
            )}ms\x1b[0m\n`,
          );
          break;
        }

        case "chunk": {
          if (inReasoning) {
            process.stderr.write("\n \x1b[0m\n");
            inReasoning = false;
          }
          if (!printedHeader) {
            output.write("\n ai> ");
            printedHeader = true;
          }
          output.write(String(frame.text));
          break;
        }

        case "error": {
          process.stderr.write(
            `\n \x1b[31m[error] ${String(frame.message)}\x1b[0m\n`,
          );
          break;
        }

        case "done": {
          if (inReasoning) process.stderr.write("\n \x1b[0m");
          output.write("\n");
          break;
        }

        default:
          break;
      }
    }

    const result = await turn.done;
    activeAbort = null;
    if (result.cached) {
      process.stderr.write("\n \x1b[2m(served from semantic cache)\x1b[0m\n");
    }
  }
} finally {
  rl.close();
  await agent.close();
}
```

## Expose streaming over HTTP servers

When `runs` is enabled on `createAgent()`, stream the registered `RunHandle` through `streamFromRun()` so every frame carries a monotonic `id:` cursor and `{ type: "init", conversationId, runId }` from byte 1; fall back to `framesToStream(turn.frames)` for non-run agents, and attach SSE headers with `setSseHeaders()`.

::tabs
:::tabs-item{icon="i-lucide-globe" label="Web Standard (Next.js / Hono / Nitro)"}
Works out of the box anywhere the Fetch API `Request`/`Response` contract is native (Next.js App Router, Hono, Nuxt Nitro, Bun, Deno):

```ts [server/api/chat.post.ts]
import {
  framesToStream,
  getConfiguration,
  getRun,
  setSseHeaders,
  streamFromRun,
  type ChatTurnResult,
} from "@riavzon/ai";
import { supportAgent } from "../../src/agent";

export async function POST(req: Request): Promise<Response> {
  const body = (await req.json()) as {
    userId: string;
    conversationId?: number;
    message: string;
    orgId: string;
  };

  // Mint runId up front so the client receives it immediately via X-Run-Id
  const runId = crypto.randomUUID();
  const controller = new AbortController();

  const turn = await supportAgent.chat({
    userId: body.userId,
    conversationId: body.conversationId,
    message: body.message,
    runId,
    meta: { orgId: body.orgId, region: "us-east" },
    signal: controller.signal,
  });

  // When `runs` is enabled on createAgent(), stream through `streamFromRun()`
  // so every frame carries a monotonic `id:` cursor and client disconnects only
  // detach without aborting the durable turn. Fall back to `framesToStream()`
  // for stateless / non-run agents.
  const run = getRun(runId);
  if (!run) {
    req.signal.addEventListener("abort", () => controller.abort(), {
      once: true,
    });
  }

  const sseStream = run
    ? streamFromRun({
        run,
        conversationId: run.conversationId,
        runId: run.id,
        log: getConfiguration().logger,
        heartbeatMs: 10_000,
      })
    : framesToStream(turn.frames, {
        heartbeatMs: 10_000,
        onCancel: () => controller.abort(),
      });

  // Optional server-side logging when the turn settles
  void turn.done.then((res: ChatTurnResult) => {
    if (res.error) console.error("Turn failed:", res.error);
  });

  return new Response(sseStream, {
    headers: {
      ...setSseHeaders(),
      "X-Run-Id": runId,
    },
  });
}
```

:::

:::tabs-item{icon="i-lucide-server" label="Node.js (Express / Fastify / node:http)"}
Pipe the `ReadableStream<Uint8Array>` chunk by chunk into `ServerResponse`, flushing headers immediately and detaching cleanly on `req.on("close")`:

```ts [server/routes/chat.ts]
import type { IncomingMessage, ServerResponse } from "node:http";
import {
  framesToStream,
  getConfiguration,
  getRun,
  setSseHeaders,
  streamFromRun,
} from "@riavzon/ai";
import { supportAgent } from "../../src/agent";

export async function handleChat(
  req: IncomingMessage,
  res: ServerResponse,
  body: {
    userId: string;
    conversationId?: number;
    message: string;
    orgId: string;
  },
): Promise<void> {
  const runId = crypto.randomUUID();
  const controller = new AbortController();

  const turn = await supportAgent.chat({
    userId: body.userId,
    conversationId: body.conversationId,
    message: body.message,
    runId,
    meta: { orgId: body.orgId, region: "us-east" },
    signal: controller.signal,
  });

  const headers = {
    ...setSseHeaders(),
    "X-Run-Id": runId,
  };
  res.writeHead(200, headers);
  res.flushHeaders();

  const run = getRun(runId);
  const sseStream = run
    ? streamFromRun({
        run,
        conversationId: run.conversationId,
        runId: run.id,
        log: getConfiguration().logger,
        heartbeatMs: 10_000,
      })
    : framesToStream(turn.frames, {
        heartbeatMs: 10_000,
      });

  const reader = sseStream.getReader();
  let clientGone = false;

  req.on("close", () => {
    clientGone = true;
    void reader.cancel();
    // Only abort non-run turns on disconnect; durable runs stay alive for reattach
    if (!run) {
      controller.abort();
    }
  });

  try {
    while (!clientGone) {
      const { value, done } = await reader.read();
      if (done) break;
      if (!res.write(value)) {
        await new Promise<void>((resolve) => res.once("drain", resolve));
      }
    }
  } finally {
    reader.releaseLock();
    if (!res.writableEnded) res.end();
  }
}
```

### Reattach, Stop, and Retry companion routes

Pair your chat endpoint with three thin run-control routes (full details in [Event Store & Reattach](/runs/event-store-and-reattach)):

```ts [server/api/runs.ts]
import {
  FIRST_ID,
  getConfiguration,
  getRun,
  getRunById,
  requestAbort,
  retryRun,
  setSseHeaders,
  streamFromRun,
  streamFromStore,
} from "@riavzon/ai";

// GET /api/runs/:id/stream — resume from Last-Event-ID on any replica
export async function GET_stream(
  req: Request,
  runId: string,
): Promise<Response> {
  const since =
    req.headers.get("last-event-id") ??
    new URL(req.url).searchParams.get("since") ??
    FIRST_ID;
  const log = getConfiguration().logger;

  const local = getRun(runId);
  if (local) {
    return new Response(
      streamFromRun({
        run: local,
        since,
        conversationId: local.conversationId,
        runId: local.id,
        log,
      }),
      { headers: setSseHeaders() },
    );
  }

  const row = await getRunById(runId);
  if (!row) return new Response("Run not found", { status: 404 });

  return new Response(
    streamFromStore({
      runId,
      since,
      conversationId: row.conversation_id ?? undefined,
      log,
      isRunAlive: async () => {
        const latest = await getRunById(runId);
        return latest?.status === "running" || latest?.status === "interrupted";
      },
    }),
    { headers: setSseHeaders() },
  );
}

// POST /api/runs/:id/stop — abort locally + mark DB row so remote replicas stop on next heartbeat
export async function POST_stop(runId: string): Promise<Response> {
  getRun(runId)?.controller.abort();
  await requestAbort(runId);
  return Response.json({ ok: true });
}

// POST /api/runs/:id/retry — restart a failed, aborted, or interrupted run via its registered resumer
export async function POST_retry(runId: string): Promise<Response> {
  const outcome = await retryRun(runId);
  return Response.json(outcome, { status: outcome.ok ? 200 : 409 });
}
```

::note
`retryRun()` looks up a resumer registered via `registerRunResumer()` for the run's `(kind, name)`. If no resumer is registered (for example, a standalone `createAgent()` setup without a custom resumer), `POST /api/runs/:id/retry` returns `409` with `{ ok: false, reason: "no_runner" }`. See [Registering a Workflow Runner](/runs/registering-a-workflow-runner) to wire durable resumers.
::

## Consume SSE streams in a browser or frontend client

`POST /api/chat` sends a JSON body, which browser `EventSource` cannot do, so read the frames from `fetch().body` yourself. One small transport module handles parsing, `Last-Event-ID` resume, Stop and Retry; the React and Vue tabs are thin wrappers over it.

::tabs
:::tabs-item{icon="i-lucide-wrench" label="Shared Client Utils"}
Framework-agnostic transport. Every consumer tab below imports it.

```ts [client/aiStreamClient.ts]
import type { RetryOutcome, SseFrame } from "@riavzon/ai";

export interface ChatState {
  conversationId?: number;
  runId?: string;
  lastEventId?: string;
  text: string;
  reasoning: string;
  tools: { id: string; name: string; callIndex: number; callTotal: number }[];
  status: "idle" | "streaming" | "done" | "error";
  error?: string;
}

export const initialState = (conversationId?: number): ChatState => ({
  conversationId,
  text: "",
  reasoning: "",
  tools: [],
  status: "idle",
});

/** Fold one wire frame into the UI state. `SseFrame` keeps every field `unknown`. */
export function applyFrame(
  prev: ChatState,
  frame: SseFrame,
  eventId?: string,
): ChatState {
  const next: ChatState = { ...prev, lastEventId: eventId ?? prev.lastEventId };

  switch (frame.type) {
    case "init":
      if (typeof frame.conversationId === "number")
        next.conversationId = frame.conversationId;
      if (typeof frame.runId === "string") next.runId = frame.runId;
      break;
    case "chunk":
      next.text = prev.text + String(frame.text ?? "");
      break;
    case "reasoning":
      next.reasoning = prev.reasoning + String(frame.delta ?? "");
      break;
    case "tool_start":
      next.tools = [
        ...prev.tools.filter((t) => t.id !== frame.id),
        {
          id: String(frame.id),
          name: String(frame.name),
          callIndex: Number(frame.callIndex ?? 1),
          callTotal: Number(frame.callTotal ?? 1),
        },
      ];
      break;
    case "tool_end":
      next.tools = prev.tools.filter((t) => t.id !== frame.id);
      break;
    case "error":
      next.status = "error";
      next.error = String(frame.message ?? "Unknown error");
      next.tools = [];
      break;
    case "done":
      // agent.chat() always emits a trailing `done` frame, even after an
      // `error` frame, so preserve the failure state and its Retry button.
      next.status = prev.status === "error" ? "error" : "done";
      next.tools = [];
      break;
    default:
      // workflow_start / workflow_progress / agent_progress — render your own UI.
      break;
  }

  return next;
}

/** Minimal SSE reader: tolerates CRLF, skips `: heartbeat` comments, echoes `id:`. */
async function* readSse(
  body: ReadableStream<Uint8Array>,
): AsyncGenerator<{ frame: SseFrame; id?: string }> {
  const reader = body.getReader();
  const decoder = new TextDecoder();
  let buffer = "";

  try {
    for (;;) {
      const { value, done } = await reader.read();
      if (done) return;
      buffer = (buffer + decoder.decode(value, { stream: true })).replace(
        /\r\n/g,
        "\n",
      );

      const blocks = buffer.split("\n\n");
      buffer = blocks.pop() ?? "";

      for (const block of blocks) {
        let id: string | undefined;
        const data: string[] = [];

        for (const line of block.split("\n")) {
          if (!line || line.startsWith(":")) continue;
          if (line.startsWith("id:")) id = line.slice(3).trim();
          else if (line.startsWith("data:"))
            data.push(line.slice(5).trimStart());
        }

        if (data.length > 0) {
          yield { frame: JSON.parse(data.join("\n")) as SseFrame, id };
        }
      }
    }
  } finally {
    reader.releaseLock();
  }
}

/** Send a turn, reattaching once through `GET /api/runs/:id/stream` if the socket drops. */
export async function sendChat(
  body: {
    userId: string;
    orgId: string;
    conversationId?: number;
    message: string;
  },
  onState: (state: ChatState) => void,
  signal?: AbortSignal,
): Promise<ChatState> {
  let state: ChatState = {
    ...initialState(body.conversationId),
    status: "streaming",
  };
  const commit = (next: ChatState) => {
    state = next;
    onState(next);
  };
  commit(state);

  let dropped: unknown;
  try {
    const res = await fetch("/api/chat", {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify(body),
      signal,
    });

    const runId = res.headers.get("X-Run-Id");
    if (runId) commit({ ...state, runId });

    if (!res.ok || !res.body) throw new Error(`HTTP ${res.status}`);

    let terminal = false;
    for await (const { frame, id } of readSse(res.body)) {
      if (frame.type === "done" || frame.type === "error") terminal = true;
      commit(applyFrame(state, frame, id));
    }

    if (terminal || signal?.aborted) return state;
  } catch (err) {
    if (signal?.aborted) return state;
    dropped = err;
  }

  if (state.runId) {
    return attachRun(state.runId, state, onState, signal);
  }

  commit({
    ...state,
    status: "error",
    error: dropped instanceof Error ? dropped.message : String(dropped),
  });
  return state;
}

/** Replay + follow a run from its event log, resuming after the last seen cursor. */
export async function attachRun(
  runId: string,
  prev: ChatState,
  onState: (state: ChatState) => void,
  signal?: AbortSignal,
): Promise<ChatState> {
  // Without a cursor the server replays from the first event, so drop the
  // buffers; with one it only sends what is new and the buffers must stay.
  let state: ChatState = prev.lastEventId
    ? { ...prev, runId, status: "streaming", error: undefined }
    : { ...initialState(prev.conversationId), runId, status: "streaming" };
  onState(state);

  const headers: Record<string, string> = prev.lastEventId
    ? { "Last-Event-ID": prev.lastEventId }
    : {};

  try {
    const res = await fetch(`/api/runs/${encodeURIComponent(runId)}/stream`, {
      headers,
      signal,
    });
    if (!res.ok || !res.body) throw new Error(`HTTP ${res.status}`);

    for await (const { frame, id } of readSse(res.body)) {
      state = applyFrame(state, frame, id);
      onState(state);
    }
  } catch (err) {
    if (signal?.aborted) return state;
    state = {
      ...state,
      status: "error",
      error: err instanceof Error ? err.message : String(err),
      tools: [],
    };
    onState(state);
  }

  return state;
}

export async function stopRun(runId: string): Promise<void> {
  const res = await fetch(`/api/runs/${encodeURIComponent(runId)}/stop`, {
    method: "POST",
  });
  if (!res.ok) throw new Error(`Failed to stop run: HTTP ${res.status}`);
}

export async function retryRun(runId: string): Promise<RetryOutcome> {
  const res = await fetch(`/api/runs/${encodeURIComponent(runId)}/retry`, {
    method: "POST",
  });
  return (await res.json()) as RetryOutcome;
}
```

```ts [hooks/useAgentChat.ts]
import { useCallback, useEffect, useRef, useState } from "react";
import type { RetryOutcome } from "@riavzon/ai";
import {
  attachRun,
  initialState,
  retryRun,
  sendChat,
  stopRun,
  type ChatState,
} from "../client/aiStreamClient";

export function useAgentChat(userId: string, orgId: string) {
  const [state, setState] = useState<ChatState>(() => initialState());
  const latest = useRef(state);
  const abort = useRef<AbortController | null>(null);

  const commit = useCallback((next: ChatState) => {
    latest.current = next;
    setState(next);
  }, []);

  useEffect(() => () => abort.current?.abort(), []);

  const send = useCallback(
    async (message: string) => {
      if (!message.trim()) return;
      abort.current?.abort();
      const controller = new AbortController();
      abort.current = controller;
      await sendChat(
        {
          userId,
          orgId,
          conversationId: latest.current.conversationId,
          message,
        },
        commit,
        controller.signal,
      );
    },
    [userId, orgId, commit],
  );

  const stop = useCallback(async () => {
    const runId = latest.current.runId;
    abort.current?.abort();
    abort.current = null;
    try {
      if (runId) await stopRun(runId);
    } finally {
      const prev = latest.current;
      commit({
        ...prev,
        status: prev.status === "streaming" ? "done" : prev.status,
        tools: [],
      });
    }
  }, [commit]);

  const retry = useCallback(async () => {
    const runId = latest.current.runId;
    if (!runId) return;

    let outcome: RetryOutcome;
    try {
      outcome = await retryRun(runId);
    } catch (err) {
      commit({ ...latest.current, status: "error", error: String(err) });
      return;
    }
    if (!outcome.ok) {
      commit({ ...latest.current, status: "error", error: outcome.message });
      return;
    }

    abort.current?.abort();
    const controller = new AbortController();
    abort.current = controller;
    await attachRun(runId, latest.current, commit, controller.signal);
  }, [commit]);

  return {
    state,
    isStreaming: state.status === "streaming",
    canRetry: state.status === "error" && Boolean(state.runId),
    send,
    stop,
    retry,
  };
}
```

::note
**Wire details the transport handles for you:**

- **Heartbeats:** `framesToStream()`, `streamFromRun()`, `streamFromStore()` and `workflowStream()` write `: heartbeat\n\n` on idle; `readSse()` skips any line starting with `:`.
- **Cursors:** `framesToStream(turn.frames)` writes bare `data:` blocks, while `streamFromRun()` and `streamFromStore()` prefix each stored event with `id: <cursor>`. Only the latter support incremental `Last-Event-ID` resume.
- **Terminal frames:** `agent.chat()` always pushes `{ type: "done" }` in `finally`, even after `{ type: "error" }`; `applyFrame()` keeps `status: "error"` so the Retry button survives.
- **Retry:** `POST /api/runs/:id/retry` answers `200` with `{ ok: true, runId, kind, name, resumed, attempts }` or `409` with `{ ok: false, reason, message }`, and resumes in the background. Reattaching with the last seen `Last-Event-ID` skips the failed attempt's `error` frame and tails only the new one.
